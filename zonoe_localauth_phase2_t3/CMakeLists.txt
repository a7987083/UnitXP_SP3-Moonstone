cmake_minimum_required(VERSION 3.16)
project(ZonoeLocalAuthPhase2T3 LANGUAGES C CXX OBJC OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(ZonoeLocalAuthPhase2T3 SHARED src/ZonoeAuthAdapter.mm)
target_include_directories(ZonoeLocalAuthPhase2T3 PRIVATE include)
set_target_properties(ZonoeLocalAuthPhase2T3 PROPERTIES OUTPUT_NAME "ZonoeLocalAuth_Phase2_T3" PREFIX "" SUFFIX ".dylib" XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED "NO" XCODE_ATTRIBUTE_CODE_SIGNING_REQUIRED "NO" XCODE_ATTRIBUTE_ENABLE_BITCODE "NO")
target_compile_options(ZonoeLocalAuthPhase2T3 PRIVATE -fobjc-arc)
target_link_libraries(ZonoeLocalAuthPhase2T3 PRIVATE "-framework Foundation" "-lobjc")
